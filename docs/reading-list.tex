\section*{Start here}
\begin{itemize}
  \item \textbf{The Vehicle tutorial.} Chapters: introduction; the language (ACAS~Xu); proving robustness (MNIST); property-driven training; exporting to theorem provers; \emph{Vancomycert}. The afternoon exercise is adapted from chapters 2–4. \url{https://vehicle-lang.github.io/tutorial/}
  \item \textbf{Tutorial chapter 4, property-driven training} — a Fashion-MNIST classifier at 99.9\,\% training accuracy has only 21–25 of 50 images provably robust at $\varepsilon = 0.005$; accuracy is not safety. \url{https://vehicle-lang.github.io/tutorial/chapters/property-driven-training/}
  \item \textbf{Vehicle documentation and source} — reference docs \url{https://vehicle-lang.readthedocs.io/en/latest/}; compiler (BSD-3-Clause) \url{https://github.com/vehicle-lang/vehicle}; tutorial repository, branch \texttt{exercises}, \url{https://github.com/vehicle-lang/tutorial}
  \item \textbf{Tutorial provenance} — run at FoMLAS@CAV 2023, the VeTSS Summer School 2023, ICFP and the SAT/SMT Summer School; tutorial paper on EasyChair: \url{https://easychair.org/publications/download/Rkrv}
  \item Cordeiro, Daggitt, Girard-Satabin, Isac, Johnson, Katz, Komendantskaya, Lemesle, Manino, Šinkarovs, Wu. \emph{Neural Network Verification is a Programming Language Challenge}. ESOP 2025. arXiv:2501.05867 — why specification languages, and the embedding gap.
  \item Kaulen, Ladner, Bak, Brix, Duong, Flinkow, Johnson, Koller, Manino, Nguyen, Wu. \emph{The 6th International Verification of Neural Networks Competition (VNN-COMP 2025)}. arXiv:2512.19007 — the ecosystem: ONNX + VNN-LIB, eight tools, 25 benchmarks.
\end{itemize}
\section*{Verification: foundations and tools}
\begin{itemize}
  \item Katz, Barrett, Dill, Julian, Kochenderfer. \emph{Reluplex: An Efficient SMT Solver for Verifying Deep Neural Networks}. CAV 2017. arXiv:1702.01135 — the ACAS~Xu benchmark, NP-completeness, and the case-splitting idea behind Marabou.
  \item Wu et al. \emph{Marabou 2.0: A Versatile Formal Analyzer of Neural Networks}. arXiv:2401.14461 — the solver used today.
  \item Daggitt, Kokke, Atkey, Arnaboldi, Slusarz, Komendantskaya et al. \emph{Vehicle: Interfacing Neural Network Verifiers with Interactive Theorem Provers}. arXiv:2202.05207; and \emph{Bridging the Embedding Gap in the Verification of Neuro-Symbolic Programs}, FSCD 2025 (arXiv:2401.06379) — the car controller that provably stays on the road.
  \item Athavale et al. \emph{Verifying Global Two-Safety Properties in Neural Networks with Confidence}. CAV 2024 — self-composition (the twin network) for monotonicity and fairness.
\end{itemize}
\section*{Adversarial examples}
\begin{itemize}
  \item Szegedy et al. \emph{Intriguing properties of neural networks}. 2013. arXiv:1312.6199 — the first adversarial examples: imperceptible noise flips the label.
  \item Goodfellow, Shlens, Szegedy. \emph{Explaining and Harnessing Adversarial Examples}. 2014. arXiv:1412.6572 — the panda that became a gibbon.
  \item Eykholt et al. \emph{Robust Physical-World Attacks on Deep Learning Visual Classification} (CVPR 2018; arXiv:1707.08945) — stickers on a stop sign read as a speed limit.
  \item \emph{Revisiting Physical-World Adversarial Attack on Traffic Sign Recognition: A Commercial Systems Perspective}. NDSS 2025. arXiv:2409.09860 — the stickers still work on commercial systems.
  \item Finlayson et al. \emph{Adversarial attacks on medical machine learning}. \emph{Science} 363:1287, 2019 — medical images are attackable too.
  \item Sharif et al. \emph{Accessorize to a Crime}. CCS 2016 — adversarial eyeglass frames make you someone else to a face recogniser.
\end{itemize}
